//== Kafra Ragnhild (Payon) - service de teleportation, en premier =========
payon,164,233,3	script	Kafra Ragnhild#reign	4_F_KAFRA1,{
	mes "[Kafra Ragnhild]";
	mes "Service Kafra de Payon. On ne voit pas beaucoup de monde ici, mais ceux qui passent repartent toujours satisfaits.";
	next;
	switch (select("Service de teleportation.:Acceder a l'entrepot.:Fixer mon point de sauvegarde ici.:Discuter un peu.:Rien, merci.")) {
	case 1:
		mes "[Kafra Ragnhild]";
		mes "Izlude, Alberta, Morocc... la foret ne mene nulle part toute seule. Ou voulez-vous aller ?";
		next;
		switch (select("Izlude (3600z).:Alberta (3200z).:Morocc (8700z).:Laisser tomber.")) {
		case 1: .@map$ = "izlude"; .@x = 128; .@y = 246; .@cost = 3600; break;
		case 2: .@map$ = "alberta"; .@x = 192; .@y = 147; .@cost = 3200; break;
		case 3: .@map$ = "morocc"; .@x = 159; .@y = 93; .@cost = 8700; break;
		case 4:
			mes "[Kafra Ragnhild]";
			mes "Une autre fois, alors.";
			close;
		}
		if (Zeny < .@cost) {
			mes "[Kafra Ragnhild]";
			mes "Il vous manque des zeny pour ce trajet.";
			close;
		}
		Zeny -= .@cost;
		mes "[Kafra Ragnhild]";
		mes "Voila. Bon voyage - la foret sera toujours la a votre retour.";
		close2;
		warp .@map$,.@x,.@y;
		end;
	case 2:
		mes "[Kafra Ragnhild]";
		mes "Bien sur. Voici votre entrepot.";
		close2;
		openstorage;
		end;
	case 3:
		mes "[Kafra Ragnhild]";
		mes "C'est fait. Payon sera votre port d'attache, desormais.";
		close2;
		savepoint "payon",164,233;
		end;
	case 4:
		mes "[Kafra Ragnhild]";
		mes "Payon n'a jamais vraiment choisi de camp, vous savez. Ni l'Ordre, ni les Forgenoires. Mais on en voit de plus en plus trainer aux abords de la ville, ces derniers temps.";
		next;
		mes "[Kafra Ragnhild]";
		mes "Ca ne nous protegera pas eternellement, je sais. J'ai bien peur qu'on doive finir par choisir, tot ou tard.";
		close;
	case 5:
		mes "[Kafra Ragnhild]";
		mes "Bonne route.";
		close;
	}
}

//== Gudrun (Payon) - identification d'objets, 100z par objet ==============
payon,152,182,3	script	Gudrun#reign	4_F_02,{
	mes "[Gudrun]";
	mes "Un oeil pour ce que vous trainez sans savoir ce que c'est ? 100 zeny par objet, pas plus, pas moins.";
	.@count = identifyall(false);
	if (.@count == 0) {
		next;
		mes "[Gudrun]";
		mes "Rien a identifier dans votre sac. Revenez quand vous aurez trouve quelque chose d'interessant.";
		close;
	}
	.@cost = .@count * 100;
	next;
	mes "[Gudrun]";
	mes .@count + " objet(s) non identifie(s). Ca fera " + .@cost + " zeny pour tout.";
	next;
	if (select("Payer " + .@cost + " zeny.:Non merci.") == 2) {
		mes "[Gudrun]";
		mes "Comme vous voulez. Ils resteront des mysteres.";
		close;
	}
	if (Zeny < .@cost) {
		mes "[Gudrun]";
		mes "Il vous manque des zeny.";
		close;
	}
	Zeny -= .@cost;
	identifyall(true);
	mes "[Gudrun]";
	mes "Voila. Plus aucun mystere dans ce sac.";
	close;
}

//== Kolgrim (Payon) - forgeron, raffinage (Refine UI natif) + materiaux =====
//= Le "Ameliorer" ouvre l'UI de raffinage native (refineui, cf. Vestri en
//= vanilla). Le "Acheter des materiaux" ouvre une boutique invisible
//= (callshop, view -1, jamais placee sur la carte) vendant les minerais
//= Enrichis/HD et le Blacksmith Blessing necessaires aux paliers eleves -
//= accessible a n'importe quel joueur, aucune restriction de metier :
//= callshop/le systeme d'achat natif ne verifie jamais la classe du
//= personnage, seulement son inventaire/poids et ses zeny.
payon,141,174,3	script	Kolgrim#reign	4_M_DWARF,{
	mes "[Kolgrim]";
	mes "Le meilleur forgeron a l'est de Prontera, et je ne dis pas ca pour vendre. Montrez-moi ce qu'il faut ameliorer, ou dites-moi ce qu'il vous faut.";
	next;
	switch (select("Ameliorer un equipement.:Acheter des materiaux.:Rien, merci.")) {
	case 1:
		if (getbattleflag("feature.refineui")) {
			close2;
			refineui();
			end;
		}
		mes "[Kolgrim]";
		mes "L'atelier est ferme pour l'instant. Revenez plus tard.";
		close;
	case 2:
		mes "[Kolgrim]";
		mes "Oridecon, Elunium, leurs versions Enrichies, du minerai HD pour ceux qui n'aiment pas le risque, et de quoi satisfaire un Blacksmith Blessing. Pas besoin d'etre du metier pour acheter, servez-vous.";
		close2;
		callshop "Materiaux de Kolgrim#reign",1;
		end;
	case 3:
		mes "[Kolgrim]";
		mes "Comme vous voulez.";
		close;
	}
}
