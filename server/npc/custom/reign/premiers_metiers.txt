//== Sigrun — formatrice Mage (iz_ac01, près de Kell) =======================
iz_ac01,55,80,3	script	Sigrun#reign	1_M_WIZARD,{
	if (Class != Job_Novice) {
		mes "[Sigrun]";
		mes "Tu as déjà trouvé ta voie. Je n'ai plus rien à t'apprendre ici.";
		close;
	}
	if (isbegin_quest(91004) < 2) {
		mes "[Sigrun]";
		mes "Inscris-toi d'abord auprès de la réceptionniste, en bas.";
		close;
	}
	if (JobLevel < 10) {
		mes "[Sigrun]";
		mes "Reviens quand tu auras atteint le niveau 10 de Novice (ta formation, pas ton niveau de personnage). Bram et Sunna t'y mèneront.";
		close;
	}
	mes "[Sigrun]";
	mes "Kell parle beaucoup de l'Ordre de la Faille. Moi, je préfère montrer.";
	next;
	mes "[Sigrun]";
	mes "Canaliser l'énergie brute sans passer par la Foi, ce n'est pas un don — c'est une discipline. On apprend à écouter ce qui frémit sous la peau du monde depuis la Rupture.";
	next;
	mes "[Sigrun]";
	mes "Si tu acceptes, tu quittes la voie de Novice et deviens officiellement Mage — ce n'est pas rien, c'est un vrai choix. Tu es prêt ?";
	next;
	if (select("Pas encore.", "Oui, je veux devenir Mage.") == 1) {
		mes "[Sigrun]";
		mes "Prends ton temps.";
		close;
	}
	jobchange Job_Mage;
	atcommand "@joblvlup 4";
	getitem 1601,1;// Rod
	mes "[Sigrun]";
	mes "Voilà. Un bâton, pour commencer — le reste s'apprend en le tenant, pas en en parlant.";
	close;
}
iz_ac01_a,55,80,3	duplicate(Sigrun#reign)	Sigrun#reign_a	1_M_WIZARD
