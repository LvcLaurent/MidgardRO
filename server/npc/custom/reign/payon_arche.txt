//===== Reign of Midgard ===================================================
//= Village Archer (pay_arche) - Kafra, benediction, chaine de la grotte
//===== Description: =======================================================
//= Extension de Payon vers son village satellite. Kafra sans service de
//= teleportation (assume comme un choix du village, pas un manque), meme
//= benediction de depart que Payon/Izlude recadree pour les archers, et une
//= chaine de 3 quetes (l'entree de la grotte/donjon de Payon, PAS des
//= "catacombes" - correction explicite) avec Vigdis - ton sombre/dramatique,
//= contraste volontaire avec le registre plus leger de Steinar (meme motif
//= post-Rupture, voir lore section 13). Cadrage important : il s'agit de
//= contenir ce qui vit dans la grotte, pas de monter en niveau/difficulte -
//= les Poporing du 3e tour sont au meme niveau que le reste, pas "plus bas".
//= Sert de mise en place pour un futur event d'invasion de zombies a Payon

//== Kafra Astrid (Village Archer) - pas de teleportation ==================
pay_arche,41,137,3	script	Kafra Astrid#reign	4_F_KAFRA1,{
	mes "[Kafra Astrid]";
	mes "Service Kafra du village archer. Pas de teleportation depuis ce comptoir - on n'a jamais eu assez de passage pour le justifier. Entrepot et point de sauvegarde, en revanche, oui.";
	next;
	switch (select("Acceder a l'entrepot.:Fixer mon point de sauvegarde ici.:Discuter un peu.:Rien, merci.")) {
	case 1:
		mes "[Kafra Astrid]";
		mes "Bien sur. Voici votre entrepot.";
		close2;
		openstorage;
		end;
	case 2:
		mes "[Kafra Astrid]";
		mes "C'est fait. Le village archer sera votre port d'attache, desormais.";
		close2;
		savepoint "pay_arche",41,137;
		end;
	case 3:
		mes "[Kafra Astrid]";
		mes "On m'a propose un comptoir de teleportation, une fois. J'ai refuse - ici, les gens restent. Ils ne font que passer par Payon meme.";
		close;
	case 4:
		mes "[Kafra Astrid]";
		mes "Bonne route.";
		close;
	}
}

//== Benediction du Village Archer - benediction de depart, 100 zeny =======
pay_arche,45,150,3	script	Benediction du Village Archer#reign	1_F_MERCHANT_01,{
	mes "[Benediction du Village Archer]";
	mes "Un rituel de tir avant de partir en chasse ? Main sure, oeil clair, souffle regulier. 100 zeny, et la corde ne tremble plus.";
	next;
	switch (select("Payer 100 zeny.:Ca marche vraiment ?:Non merci.")) {
	case 1:
		if (Zeny < 100) {
			mes "[Benediction du Village Archer]";
			mes "100 zeny. Meme la precision se paie, par ici.";
			close;
		}
		Zeny -= 100;
		sc_start SC_BLESSING,240000,10;
		sc_start SC_INCREASEAGI,240000,10;
		mes "[Benediction du Village Archer]";
		mes "Voila. Quatre minutes de main sure et de bonne fortune. Visez juste.";
		close;
	case 2:
		mes "[Benediction du Village Archer]";
		mes "Un vieux rituel d'archer, transmis de maitre a eleve depuis des generations. Ca marche, ou alors on y croit assez fort pour que ca revienne au meme.";
		close;
	case 3:
		mes "[Benediction du Village Archer]";
		mes "Comme vous voulez. La corde attendra.";
		close;
	}
}

//== Vigdis (Village Archer) - chaine de la grotte de Payon (91203-91205) ===
//= Cadrage : elle surveille l'entree de la grotte/donjon de Payon (pas des
//= "catacombes"). Les tours de garde sont une contention necessaire, pas une
//= escalade de difficulte - ne jamais dire que les monstres montent en
//= niveau. Le 3e tour (Poporing) est au meme niveau que le reste, on ne
//= s'enfonce pas plus loin pour l'instant.
pay_arche,55,125,3	script	Vigdis#reign	4_F_02,{
	if (isbegin_quest(91206) == 1) {
		mes "[Vigdis]";
		mes "Un paquet ? *Elle le prend, le retourne entre ses mains, et un instant, son visage change - reconnaissance, peut-etre, ou quelque chose de plus vieux que ca.*";
		next;
		mes "[Vigdis]";
		mes "Toralf. Ca faisait longtemps. Dites-lui, la prochaine fois que vous le croisez, que la garde tient toujours. Comme avant.";
		close2;
		getexp 27070,0;
		add_reputation_points(REPUTATION_PAYON,50);
		completequest 91206;
		end;
	}
	if (payon_grotte_stage >= 3) {
		mes "[Vigdis]";
		mes "L'entree de la grotte est calme, pour l'instant. Trois tours de garde montes, trois tenus. Ca ne durera pas eternellement - rien ne dure, avec ce qui vit la-dedans.";
		close;
	}

	if (payon_grotte_stage == 0) {
		if (checkquest(91203,HUNTING) == 2) {
			mes "[Vigdis]";
			mes "Dix morts-vivants de moins a remuer la terre. Le premier tour de garde est tenu.";
			next;
			mes "[Vigdis]";
			mes "Ca ne les empechera pas de revenir, mais au moins l'entree respire un peu mieux pour un moment.";
			close2;
			getexp 27070,0;
			add_reputation_points(REPUTATION_PAYON,100);
			erasequest 91203;
			payon_grotte_stage = 1;
			end;
		} else if (isbegin_quest(91203) == 1) {
			mes "[Vigdis]";
			mes "Le premier tour de garde n'est pas tenu. Dix morts-vivants remuent encore a l'entree de la grotte.";
			close;
		}
		mes "[Vigdis]";
		mes "*Elle garde les yeux fixes sur l'entree de la grotte, une main sur la garde de son epee.*";
		next;
		mes "[Vigdis]";
		mes "Depuis la Rupture, ce qui vit la-dedans est plus nombreux et plus agite. Si personne ne s'en occupe regulierement, ca finira par sortir - et ce sera Payon qui en fera les frais, pas nous.";
		next;
		mes "[Vigdis]";
		mes "Trois tours de garde, dans l'ordre. Rien d'heroique, juste ce qu'il faut pour que ca reste contenu. Le premier : dix morts-vivants, a l'entree. Vous en etes capable ?";
		next;
		switch (select("Je monte la garde.:Pas cette fois.")) {
		case 1:
			setquest 91203;
			mes "[Vigdis]";
			mes "Dix morts-vivants. Revenez me voir une fois l'entree degagee.";
			close;
		case 2:
			mes "[Vigdis]";
			mes "Comme vous voulez. La grotte attendra, elle a l'habitude.";
			close;
		}
	}

	if (payon_grotte_stage == 1) {
		if (checkquest(91204,HUNTING) == 2) {
			mes "[Vigdis]";
			mes "Squelettes et familiers repousses. Le deuxieme tour de garde est tenu.";
			next;
			mes "[Vigdis]";
			mes "Il reste un dernier tour.";
			close2;
			getexp 27070,0;
			add_reputation_points(REPUTATION_PAYON,100);
			erasequest 91204;
			payon_grotte_stage = 2;
			end;
		} else if (isbegin_quest(91204) == 1) {
			mes "[Vigdis]";
			mes "Le deuxieme tour de garde n'est pas tenu. Squelettes et familiers rodent encore plus loin dans la grotte.";
			close;
		}
		mes "[Vigdis]";
		mes "Le premier tour de garde est tenu. Plus loin dans la grotte, dix squelettes et cinq familiers rodent encore.";
		next;
		switch (select("Je continue.:Pas cette fois.")) {
		case 1:
			setquest 91204;
			mes "[Vigdis]";
			mes "Dix squelettes, cinq familiers. Ne les laissez pas vous encercler.";
			close;
		case 2:
			mes "[Vigdis]";
			mes "Comme vous voulez. Je resterai postee ici, avec ou sans vous.";
			close;
		}
	}

	if (payon_grotte_stage == 2) {
		if (checkquest(91205,HUNTING) == 2) {
			mes "[Vigdis]";
			mes "...Des Poporing ? Apres deux tours de garde contre des morts-vivants, celui-la se termine sur des Poporing.";
			next;
			mes "[Vigdis]";
			mes "Trois tours de garde tenus. Vous avez fait plus pour ce village que bien des gardes en uniforme.";
			close2;
			getexp 27070,0;
			add_reputation_points(REPUTATION_PAYON,100);
			erasequest 91205;
			payon_grotte_stage = 3;
			end;
		} else if (isbegin_quest(91205) == 1) {
			mes "[Vigdis]";
			mes "Le dernier tour de garde n'est pas tenu. Dix Poporing, au meme niveau que le reste.";
			close;
		}
		mes "[Vigdis]";
		mes "Deux tours de garde tenus. Le troisieme : des Poporing, au meme niveau que le reste. Rien de plus dangereux, juste une bete de plus a tenir a l'oeil.";
		next;
		switch (select("Je termine ce qui a commence.:Pas cette fois.")) {
		case 1:
			setquest 91205;
			mes "[Vigdis]";
			mes "Dix Poporing. Le dernier tour de garde.";
			close;
		case 2:
			mes "[Vigdis]";
			mes "Comme vous voulez. Je ne vous en voudrai pas d'hesiter.";
			close;
		}
	}
OnInit:
	questinfo QTYPE_QUEST, QMARK_YELLOW, "isbegin_quest(91206) == 1";
	questinfo QTYPE_QUEST, QMARK_YELLOW, "payon_grotte_stage == 0 && checkquest(91203,HUNTING) == 2";
	questinfo QTYPE_QUEST, QMARK_YELLOW, "payon_grotte_stage == 0 && isbegin_quest(91203) == 0";
	questinfo QTYPE_QUEST2, QMARK_NONE, "payon_grotte_stage == 0 && isbegin_quest(91203) == 1";
	questinfo QTYPE_QUEST, QMARK_YELLOW, "payon_grotte_stage == 1 && checkquest(91204,HUNTING) == 2";
	questinfo QTYPE_QUEST, QMARK_YELLOW, "payon_grotte_stage == 1 && isbegin_quest(91204) == 0";
	questinfo QTYPE_QUEST2, QMARK_NONE, "payon_grotte_stage == 1 && isbegin_quest(91204) == 1";
	questinfo QTYPE_QUEST, QMARK_YELLOW, "payon_grotte_stage == 2 && checkquest(91205,HUNTING) == 2";
	questinfo QTYPE_QUEST, QMARK_YELLOW, "payon_grotte_stage == 2 && isbegin_quest(91205) == 0";
	questinfo QTYPE_QUEST2, QMARK_NONE, "payon_grotte_stage == 2 && isbegin_quest(91205) == 1";
	end;
}
