iz_ac01,96,44,5	script	Réceptionniste#reign	4_F_01,{
	if (isbegin_quest(91004) == 2) {
		mes "[Réceptionniste]";
		mes "Bonjour, " + strcharinfo(0) + ". Toujours ravie de te revoir à l'Académie d'Izlude.";
	} else {
		mes "[Réceptionniste]";
		mes "Bienvenue à l'Académie d'Izlude. Je m'occupe de l'accueil et de l'inscription des nouveaux venus.";
	}
	next;
	while (1) {
		mes "[Réceptionniste]";
		mes "En quoi puis-je t'aider ?";
		next;
		if (isbegin_quest(91004) == 2)
			.@menu$ = "Ce qu'il y a dans l'Académie et alentour.:C'est quoi, la Rupture ?:Terminer la conversation.";
		else
			.@menu$ = "M'inscrire à l'Académie:Explications sur l'Académie et Izlude:C'est quoi, la Rupture ?:Terminer la conversation";
		.@choice = select(.@menu$);
		if (isbegin_quest(91004) == 2)
			.@choice += 1;
		switch (.@choice) {
		case 1:
			if (Class == Job_Novice) {
				if (!checkweight(1301,10)) {
					mes "- Tu es trop chargé pour continuer cette quête. -";
					close;
				}
				mes "[Réceptionniste]";
				mes "Voyons... " + strcharinfo(0) + ", c'est ça ? Je t'inscris tout de suite.";
				mes "Voici de quoi commencer. Prends-en soin, on ne redonne pas deux fois un équipement de départ.";
				setquest 91004;
				completequest 91004;
				getitem 2352,1;// Novice_Plate
				getitem 2510,1;// Novice_Hood
				getitem 2414,1;// Novice_Boots
				getitem 5055,1;// Novice_Egg_Cap
				getitem 1243,1;// Novice_Knife
				getitem 2112,1;// Novice_Guard
				getitem 569,300;// Novice_Potion
				getexp 100,0;
				next;
				mes "[Réceptionniste]";
				mes "Bienvenue parmi nous. Reviens me voir si tu as des questions — sur l'Académie, ou sur ce monde en général.";
				close;
			} else {
				mes "[Réceptionniste]";
				mes "L'inscription est réservée aux Novices. Mais tu peux tout à fait rester te renseigner.";
				close;
			}
		case 2:
			mes "[Réceptionniste]";
			mes "La première salle à gauche, c'est nos instructeurs. Ils t'apprendront les gestes de base — vas-y avant tout le reste.";
			next;
			mes "[Réceptionniste]";
			mes "À droite, tu trouveras la cantine, si une petite faim te prend. Et juste au-dessus, les archives — utile si tu cherches à en savoir plus sur ce monde.";
			next;
			mes "[Réceptionniste]";
			mes "Au-dessus des instructeurs, à gauche, tu croiseras des représentants de l'Ordre de la Faille et des Forgenoires. Ils ne mordent pas — enfin, pas littéralement.";
			next;
			mes "[Réceptionniste]";
			mes "Et tout en haut, c'est l'ancienne Église. On y donne encore cours, aujourd'hui — mais plutôt sur les livres que sur les prières.";
			next;
			mes "[Réceptionniste]";
			mes "Izlude elle-même est un port franc — ni l'Ordre de la Faille ni les Forgenoires n'y imposent leur loi. C'est voulu : aucune des deux factions ne peut se permettre de fermer ce port.";
			next;
			break;
		case 3:
			mes "[Réceptionniste]";
			mes "Ah. Ça, tout le monde le demande, tôt ou tard.";
			next;
			mes "[Réceptionniste]";
			mes "Il y a une génération, quelque chose s'est fissuré sous Midgard — les racines d'Yggdrasil, dit-on. On appelle ça simplement la Rupture.";
			next;
			mes "[Réceptionniste]";
			mes "De cette fissure sont nées deux choses : une énergie brute que certains ont appris à canaliser sans passer par la Foi — l'Ordre de la Faille. Et les vestiges d'une civilisation plus ancienne, ses machines encore actives — les Forgenoires.";
			next;
			mes "[Réceptionniste]";
			mes "Deux écoles, deux façons de refuser de prier pour obtenir de la force. La Foi a crié à l'hérésie. La guerre qui a suivi n'a jamais vraiment cessé.";
			next;
			mes "[Réceptionniste]";
			mes "Et la Rupture n'a pas libéré que ça. Des créatures qu'on croyait endormies depuis des siècles ressurgissent, un peu partout. C'est pour ça que le monde a autant besoin d'aventuriers, ces temps-ci.";
			next;
			mes "[Réceptionniste]";
			mes "Voilà. Tu en sais autant que n'importe quel enfant d'ici, maintenant.";
			next;
			break;
		case 4:
			mes "[Réceptionniste]";
			mes "Reviens quand tu veux.";
			close;
		}
