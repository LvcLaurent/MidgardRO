//===== Reign of Midgard ===================================================
//= Paliers de Job Level (post-Novice)
//===== Description: =======================================================
//= L'XP de job est desactivee cote serveur (job_exp_rate a 0) - depuis
//= toujours, le Job Level ne monte que par @joblvlup accorde par un PNJ
//= autorise (Bram/Sunna pour Novice, les 6 formateurs de premiers_metiers.txt
//= pour le premier changement de metier). Ce fichier comble la suite :
//= paliers de Job Level lies au Base Level, un par un. Palier 1 (Job 10 @
//= Base 20) et palier 2 (Job 20 @ Base 30) sont en place - paliers 3-5
//= suivront le meme gabarit plus tard (voir job_tier2, verrou sequentiel
//= pour job_tier3).
//=
//= Palier 2 : second echo ouvert par Hakon sur 1@lab (laboratoire), une
//= fois job_tier1 acquis. Purement narratif, sans combat ni quete trackee -
//= 4 "paires" de PNJ figes (chacune montrant la meme scene a deux voix,
//= peu importe lequel des deux on touche) puis une scene finale a 4 voix
//= dans le laboratoire principal, chainees par teleportation. Pre-Rupture,
//= sans faction : le Cercle de Gullveig, un petit groupe de hauts
//= dignitaires de l'ancienne Foi, y a decouvert un fragment du meme pouvoir
//= qui jaillira plus tard a la Rupture - de quoi rivaliser avec les dieux -
//= et a choisi de le garder pour lui (longevite, controle) plutot que de le
//= partager. La Foi elle-meme y est fondee dans la scene finale, comme outil
//= pense pour detourner le regard du peuple plutot que comme revelation
//= sincere. Voir lore/reign-of-midgard-lore.md section correspondante.
//=
//= UNE seule quete pour ce palier (91100), commune aux 6 premiers metiers
//= (Swordman, Mage, Acolyte, Archer, Merchant, Thief) - pas de tirage
//= aleatoire pour l'instant (a revoir plus tard).
//=
//= Cadre : Hakon le Passeur (iz_ac02, meme batiment que l'Academie, juste
//= l'etage superieur - npc/re/jobs/novice/academy.txt est commente dans
//= npc/re/scripts_jobs.conf, aucun conflit) ouvre un echo du passe sur
//= 1@spa (carte "Ghost Palace" du client, chargeable directement sans
//= passer par le systeme d'instance - confirmee presente dans
//= db/map_index.txt). Le palais appartenait a un juge de l'ancien monde,
//= Kolvar, qui a decouvert des signes avant-coureurs de la Rupture et a
//= choisi de faire taire ceux qui les avaient vus plutot que d'en parler -
//= premisses de la piste ouverte dans lore/reign-of-midgard-lore.md section 2 sur
//= la nature reelle de la Rupture. Volontairement laisse en suggestion, pas
//= en reponse.
//=
//= Pas d'instance rAthena : tout se passe sur 1@spa en carte partagee/
//= persistante, comme le champ de Geir sur prt_fild08. Ca evite le probleme
//= de mort/deconnexion qui bloquerait un joueur hors d'une instance privee.
//= La derniere phase (duel invoque, non trackee par le systeme de quete
//= natif) utilise mobcount() sur un evenement de mort partage plutot qu'un
//= compteur par joueur - dans de tres rares cas de deux joueurs sur cette
//= phase exactement en meme temps, ca peut juste faire attendre un peu plus
//= longtemps que necessaire, jamais casser quoi que ce soit.
//=
//= Progression (var permanente par personnage, sans prefixe) :
//=   job_tier1_stage  0=pas commence, 1=catacombes, 2=bibliotheque,
//=                    3=salle du trone, 4=duel invoque en cours
//=   job_tier1        0/1, passe a 1 une fois le palier complete
//=   job_tier2_stage  0=pas commence, 1=paire 1 a voir, 2=paire 2,
//=                    3=paire 3, 4=paire 4, 5=scene finale a voir
//=   job_tier2        0/1, passe a 1 une fois le palier complete
//===========================================================================

-	script	PalierJob1#reign	-1,{
	end;

OnInit:
	// Prisonniers des catacombes - garnison permanente, melange de deux
	// epoques du meme crime (zombie = plus recents, squelette = plus anciens).
	for (.@i = 0; .@i < 10; .@i++)
		monster "1@spa", 38 + rand(-10,10), 41 + rand(-10,10), "Prisonnier decompose", 1015, 1; // ZOMBIE
	for (.@i = 0; .@i < 10; .@i++)
		monster "1@spa", 38 + rand(-10,10), 41 + rand(-10,10), "Prisonnier oublie depuis longtemps", 1076, 1; // SKELETON
	end;

// Landing label pour les gardes invoques par le Juge (salle du trone) -
// aucune logique ici, juste un event a rattacher pour que mobcount() puisse
// les compter. Rien a faire au deces (voir Le Juge#reign pour la logique).
OnGuardDown:
	end;
}

// -- Recompense partagee du palier 1 -----------------------------------
// @joblvlup jusqu'a job 10 pile (jamais un nombre fixe - correct quel que
// soit le job level de depart), 10000z, un peu de reputation (meme pattern
// que Geir : faction du joueur -> sa faction seule, sinon split Ordre/Forge).
function	script	F_RecompensePalier1	{
	if (JobLevel < 10)
		atcommand "@joblvlup " + (10 - JobLevel);
	Zeny += 10000;
	if (getfaction() == 1)
		add_reputation_points(REPUTATION_ORDRE,10);
	else if (getfaction() == 2)
		add_reputation_points(REPUTATION_FORGENOIRES,10);
	else {
		add_reputation_points(REPUTATION_ORDRE,5);
		add_reputation_points(REPUTATION_FORGENOIRES,5);
	}
	job_tier1 = 1;
	job_tier1_stage = 0;
	return;
}
