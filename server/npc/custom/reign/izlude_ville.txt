//== Kafra Elin (izlude) — entrepôt + point de sauvegarde ====================
izlude,130,148,3	script	Kafra Elin#reign	4_F_KAFRA1,{
	mes "[Kafra Elin]";
	mes "Bienvenue au service Kafra d'Izlude ! Entrepôt, point de sauvegarde... que puis-je pour vous ?";
	next;
	switch (select("Service de téléportation.:Accéder à l'entrepôt.:Fixer mon point de sauvegarde ici.:Discuter un peu.:Rien, merci.")) {
	case 1:
		mes "[Kafra Elin]";
		mes "Geffen, Aldebaran, Morroc, Payon... le monde est plus grand qu'Izlude, figurez-vous. Ou voulez-vous aller ?";
		next;
		switch (select("Geffen (4200z).:Aldebaran (4200z).:Morroc (8100z).:Payon (3600z).:Laisser tomber.")) {
		case 1: .@map$ = "geffen"; .@x = 119; .@y = 59; .@cost = 4200; break;
		case 2: .@map$ = "aldebaran"; .@x = 139; .@y = 131; .@cost = 4200; break;
		case 3: .@map$ = "morocc"; .@x = 159; .@y = 93; .@cost = 8100; break;
		case 4: .@map$ = "payon"; .@x = 162; .@y = 233; .@cost = 3600; break;
		case 5:
			mes "[Kafra Elin]";
			mes "Une autre fois, alors.";
			close;
		}
		if (Zeny < .@cost) {
			mes "[Kafra Elin]";
			mes "Il vous manque des zeny pour ce trajet.";
			close;
		}
		Zeny -= .@cost;
		mes "[Kafra Elin]";
		mes "Voila. Bon voyage.";
		close2;
		warp .@map$,.@x,.@y;
		end;
	case 2:
		mes "[Kafra Elin]";
		mes "Bien sûr. Voici votre entrepôt.";
		close2;
		openstorage;
		end;
	case 3:
		mes "[Kafra Elin]";
		mes "C'est fait. Izlude sera votre port d'attache, désormais — au sens propre.";
		close2;
		savepoint "izlude",130,148;
		end;
	case 4:
		mes "[Kafra Elin]";
		mes "On en voit passer, des naufragés, depuis quelque temps. Vous n'êtes pas les premiers, et sûrement pas les derniers.";
		next;
		mes "[Kafra Elin]";
		mes "Kafra ne pose pas de questions sur d'où vous venez. On stocke vos affaires, on retient votre visage, et c'est tout ce qui compte pour nous.";
		close;
	case 5:
		mes "[Kafra Elin]";
		mes "Bon vent.";
		close;
	}
}
izlude_a,130,148,3	duplicate(Kafra Elin#reign)	Kafra Elin#reign_a	4_F_KAFRA1

//== Herjolf (izlude) — éclaireur, choix de faction (facultatif) ============
//= Utilise getfaction()/setfaction(), exposés côté serveur pour cette quête
//= (cf. commit "Add faction system"). Le choix n'est jamais forcé et, une
//= fois fait, n'est plus reproposé par ce PNJ.
izlude,44,99,3	script	Herjolf#reign	8W_SOLDIER,{
	.@faction = getfaction();
	if (.@faction == 1) {
		mes "[Herjolf]";
		mes "L'Ordre de la Faille, toujours ? Bien. Un jour, ça se voit sur un visage.";
		close;
	} else if (.@faction == 2) {
		mes "[Herjolf]";
		mes "Les Forgenoires. Ça vous va bien, ce genre de certitude.";
		close;
	}
	mes "[Herjolf]";
	mes "*Il vous jauge un instant avant de reprendre son couteau et son bout de bois.*";
	next;
	mes "[Herjolf]";
	mes "Encore un sans-faction. Ça se voit à la façon dont on vous regarde, ici. À Izlude, ça ne change rien. Ailleurs, ça peut vous coûter cher.";
	next;
	mes "[Herjolf]";
	mes "Un garde de Prontera vous laissera peut-être tranquille. Un zélote de l'Ordre ou un chasseur de primes Forgenoire, sur les routes, ailleurs ? Nettement moins sûr.";
	next;
	mes "[Herjolf]";
	mes "Je ne dis pas ça pour vous faire peur. Enfin — un peu, si. Mais surtout parce que personne ne me l'a dit, à moi, et j'ai mis longtemps à comprendre pourquoi on me regardait de travers à Geffen.";
	next;
	mes "[Herjolf]";
	mes "Alors. Vous voulez choisir un camp, maintenant ? Ou vous préférez encore réfléchir ? Je ne serai pas toujours là pour vous le redemander gentiment.";
	next;
	switch (select("L'Ordre de la Faille.:Les Forgenoires.:Pas encore.")) {
	case 1:
		setfaction 1;
		mes "[Herjolf]";
		mes "L'Ordre de la Faille, donc. Geffen se souviendra de votre nom, tôt ou tard.";
		close;
	case 2:
		setfaction 2;
		mes "[Herjolf]";
		mes "Les Forgenoires. Lighthalzen ne vous demandera jamais de croire — juste de construire.";
		close;
	case 3:
		mes "[Herjolf]";
		mes "Comme vous voulez. Ce n'est pas à moi de choisir à votre place. Juste... ne traînez pas trop hors des murs d'Izlude sans y avoir réfléchi.";
		close;
	}
}
