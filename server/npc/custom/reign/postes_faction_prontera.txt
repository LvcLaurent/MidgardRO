//===== Reign of Midgard ===================================================
//= Postes de faction - Prontera
//===== Description: =======================================================
//= Quatre PNJ lies au raid de faction sur Prontera (voir
//= raid_faction_prontera.txt, PRC_Prontera#reign) :
//= - Deux postes d'ATTAQUE, hors les murs (un par faction). Visibles
//=   seulement si ma faction NE possede PAS la ville. Lancent l'assaut pour
//=   5000z (declenchement differe de 5 minutes, cf. OnLaunchAttack) et
//=   permettent de preparer la prochaine vague d'attaque (plafond 70,
//=   depensee integralement au declenchement du raid).
//= - Deux postes de DEFENSE "Gardien", dans la ville. Visibles seulement si
//=   ma faction POSSEDE la ville. Permettent de recruter une garnison
//=   permanente (plafond 50) qui pop immediatement et reste presente meme
//=   sans raid en cours ni joueur connecte.
//= La visibilite reelle est geree par F_SyncPostesVisibility (appelee par le
//= controleur a l'init, a la fin de chaque raid et a chaque changement de
//= propriete) - disablenpc/enablenpc, pas juste un hideonnpc cosmetique.
//===========================================================================

prt_fild08,166,371,4	script	Poste de l'Ordre#reign	1_M_KNIGHTMASTER,{
	.@my_faction = 1;

	if (getfaction() != .@my_faction) {
		mes "[Poste de l'Ordre]";
		mes "Seuls les membres de l'Ordre de la Faille peuvent utiliser ce poste.";
		close;
	}

	if (getvariableofnpc(.Active, "PRC_Prontera#reign"))
		callfunc("F_AnnonceProchaineAttaque", "[Poste de l'Ordre]");

	mes "[Poste de l'Ordre]";
	mes "Prontera ne nous appartient pas. Lance l'assaut ou prepare nos troupes avant l'attaque.";
	next;
	switch (select("Lancer l'attaque (5000z)", "Preparer des renforts d'attaque", "Rien pour l'instant")) {
	case 1:
		if (Zeny < 5000) {
			mes "[Poste de l'Ordre]";
			mes "Il faut 5000z pour lancer l'assaut.";
			close;
		}
		set Zeny, Zeny - 5000;
		$@RAID_TRIGGER_FACTION = .@my_faction;
		donpcevent "PRC_Prontera#reign::OnLaunchAttack";
		mes "[Poste de l'Ordre]";
		mes "L'ordre est donne. Les troupes se rassemblent, l'assaut commencera dans 5 minutes.";
		close;
	case 2:
		callfunc("F_MenuAchatGardiens", .@my_faction, "ATK", 70);
	case 3:
		close;
	}
}

prontera,144,201,4	script	Gardien de l'Ordre#reign	1_M_KNIGHTMASTER,{

		mes "[Gardien de l'Ordre]";

		close;
