//===== Reign of Midgard ===================================================
//= Bonus de connexion quotidien
//===== Description: =======================================================
//= OnPCLoginEvent (special label, cf. doc/script_commands.txt) plutot que
//= le systeme "Attendance" natif (db/re/attendance.yml) : celui-ci ne peut
//= donner que de vrais objets par mail et demande de gerer des plages de
//= dates qui se suivent sans jamais se chevaucher. Ici, verification simple
//= via gettime(DT_YYYYMMDD) - fonctionne indefiniment, aucune plage a
//= entretenir. Donne un Jeton Bronze de la Roulette (item 90000, voir
//= db/import-tmpl/item_db.yml) plutot que d'incrementer RouletteBronze
//= directement - le joueur garde la main sur quand il l'utilise.
//= last_daily_bonus est une var persistante PAR PERSONNAGE (sans prefixe),
//= pas par compte (#) : les points de roulette eux-memes sont par
//= personnage, donc chaque personnage doit pouvoir reclamer son propre bonus.
//===========================================================================

-	script	BonusQuotidien#reign	-1,{
	end;

OnPCLoginEvent:
	if (last_daily_bonus != gettime(DT_YYYYMMDD)) {
		last_daily_bonus = gettime(DT_YYYYMMDD);
		getitem 90000,1; // Jeton_Roulette_Bronze
		dispbottom "Bonus quotidien : un Jeton Bronze de la Roulette vous attend dans votre inventaire.";
	}
	end;
}
