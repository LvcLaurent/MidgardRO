//== Instructeur Bram (iz_ac01) — Basic Skill niveaux 1 à 4 =================
iz_ac01,85,45,5	script	Instructeur Bram#reign	4_M_OLDSCHOLAR,{
	if (isbegin_quest(91004) < 2) {
		mes "[Instructeur Bram]";
		mes "Va d'abord t'inscrire auprès de la réceptionniste, on parlera ensuite.";
		close;
	}
	if (isbegin_quest(91005) == 0) {
		mes "[Instructeur Bram]";
		mes "Ah, une nouvelle recrue. Bienvenue. Je vais t'apprendre les gestes de base — tu en auras besoin bien plus souvent que tu ne le crois.";
		next;
		mes "[Instructeur Bram]";
		mes "Le commerce, d'abord. Un clic droit sur un autre aventurier te permet de lui proposer un échange d'objets.";
		next;
		mes "[Instructeur Bram]";
		mes "Ensuite, les émotions — Alt et une touche numérique, ou Alt+L pour la liste complète. Ça paraît futile, mais ça évite bien des malentendus.";
		next;
		mes "[Instructeur Bram]";
		mes "Apprends aussi à t'asseoir — la touche Inser, ou la commande /sit. Tu récupères ta santé et ton énergie bien plus vite immobile.";
		next;
		mes "[Instructeur Bram]";
		mes "Et enfin, la création de salon de discussion, avec Alt+C. Utile pour recruter, vendre, ou simplement parler sans déranger tout Izlude.";
		next;
		mes "[Instructeur Bram]";
		mes "Voilà. Quatre choses simples, mais qui changent tout au quotidien.";
		if (getskilllv(NV_BASIC) < 4) skill NV_BASIC,4,0;
		setquest 91005;
		completequest 91005;
		getexp 50,50;
		close;
	}
	mes "[Instructeur Bram]";
	mes "Va donc voir Sunna, un peu plus loin. Elle complètera ce que je t'ai appris.";
	close;

OnInit:
	questinfo( QTYPE_QUEST, QMARK_YELLOW, "isbegin_quest(91004) == 2 && isbegin_quest(91005) == 0 && Class == Job_Novice" );
	end;
}
