		mes "[Geir]";
		mes "Vingt Fabre, dix Pupa. Tu as fait le compte. Le champ respire déjà mieux.";
		next;
		mes "[Geir]";
		mes "Tiens, tu as gagné ça sur le terrain — c'est amplement mérité.";
		close2;
		getexp 20303,0;
		erasequest 91007;
		end;
	} else if (isbegin_quest(91007) == 1) {
		mes "[Geir]";
		mes "Toujours pas fini ? Ils grouillent, tu ne devrais pas avoir de mal à trouver ton compte. Vingt Fabre, dix Pupa.";
		close;
	}

	mes "Ne ris pas. Je sais de quoi ça a l'air, un garde-champêtre qui s'inquiète pour des Fabre et des Pupa.";
	next;
	mes "[Geir]";

	next;
	mes "[Geir]";
	mes "...ce qui en ressort n'a plus rien d'un ver de terre. Une sorte de papillon, mais raté, difforme. Vicieux. On en a vu deux, la semaine dernière. Ça a suffi à vider le champ de novices pendant trois jours.";

		close;
	case 2:

		mes "[Geir]";
		mes "Vingt Fabre, dix Pupa. Reviens me voir une fois que c'est fait.";
		close;
	}
