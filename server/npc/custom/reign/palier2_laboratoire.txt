//===== Reign of Midgard ===================================================
//= Palier de Job 2 (Job 20 @ Base 30) - Le Cercle de Gullveig
//===== Description: =======================================================
//= Second echo ouvert par Hakon le Passeur (paliers_job.txt), sur 1@lab.
//= Purement narratif : pas de combat, pas de quete trackee, juste une
//= progression lineaire par teleportation entre 4 "paires" de PNJ figes
//= puis une scene finale a 4 voix. Chaque paire montre EXACTEMENT la meme
//= scene a deux voix, quel que soit celui des deux PNJ qu'on touche - un
//= second PNJ (duplicate) est place a cote du premier uniquement pour
//= l'effet visuel "on tombe sur une vraie conversation".
//=
//= Pre-Rupture, sans faction (voir lore/reign-of-midgard-lore.md) : le
//= Cercle de Gullveig - Eirik, Gunnhild, Bergthor, Asa, Njal, Disa, Orm,
//= Saga - de hauts dignitaires de l'ancienne Foi, decouvre un fragment du
//= meme pouvoir qui jaillira plus tard a la Rupture, le garde pour son
//= propre profit (longevite, controle a l'echelle du divin), et finit par
//= fonder la Foi elle-meme comme outil pour detourner le regard du peuple
//= plutot que par conviction sincere.
//=
//= Progression : voir job_tier2_stage dans paliers_job.txt. Chaque scene
//= incremente le stage et televporte vers l'entree de la suivante -
//= questinfo_refresh apres chaque changement (job_tier2_stage n'est pas
//= dans la liste des declencheurs automatiques de questinfo, contrairement
//= a un changement d'etat de quete natif). ASCII pur.
//===========================================================================

// -- Recompense du palier 2 ----------------------------------------------
// Meme gabarit que F_RecompensePalier1 (paliers_job.txt) : @joblvlup jusqu'a
// job 20 pile, zeny doubles (palier superieur), meme repartition de
// reputation par faction.
function	script	F_RecompensePalier2	{
	if (JobLevel < 20)
		atcommand "@joblvlup " + (20 - JobLevel);
	Zeny += 20000;
	if (getfaction() == 1)
		add_reputation_points(REPUTATION_ORDRE,20);
	else if (getfaction() == 2)
		add_reputation_points(REPUTATION_FORGENOIRES,20);
	else {
		add_reputation_points(REPUTATION_ORDRE,10);
		add_reputation_points(REPUTATION_FORGENOIRES,10);
	}
	job_tier2 = 1;
	job_tier2_stage = 0;
	return;
}

	mes "Regarde. Ca n'a jamais cesse de battre, meme sous des siecles de pierre.";
	next;

	mes "C'est... Eirik, c'est de la lumiere qui sort de la roche elle-meme. Ca ne devrait pas etre possible.";
	next;
	mes "[Eirik]";
	mes "Rien de ce que nous enseignons n'aurait du etre possible non plus, avant qu'on ne l'ecrive. Peut-etre que c'est notre tour d'ecrire quelque chose de nouveau.";
	next;
	mes "[Gunnhild]";
	mes "Il faut prevenir le Conclave. Une decouverte comme celle-ci ne nous appartient pas.";
	next;
	mes "[Eirik]";
	mes "Le Conclave a cent mains et une seule bouche prudente. Laisse-moi au moins comprendre ce que c'est, avant qu'on nous l'arrache pour l'enfermer sans meme le regarder.";

	mes "...Une semaine. Pas plus.";
	next;

	close2;
	job_tier2_stage = 2;
	questinfo_refresh;
	warp "1@lab",97,29;
	end;
OnInit:
	questinfo QTYPE_QUEST, QMARK_YELLOW, "job_tier2_stage == 1";
	end;
}
