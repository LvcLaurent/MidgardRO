//== La folle de Prontera (prt_fild08) - collecte de Feather (LUNATIC) =======
prt_fild08,164,370,3	script	La folle de Prontera#reign	4_F_02,{
	.@feather_needed = 15;

	if (isbegin_quest(91008) == 2) {
		mes "[La folle de Prontera]";
		mes "Mon echarpe est finie. Regarde-moi cette douceur - qui aurait cru que ces sales bestioles cachaient ca sous leurs plumes.";
		close;
	}
	if (isbegin_quest(91008) == 1) {
		if (countitem(949) < .@feather_needed) {
			mes "[La folle de Prontera]";
			mes "Toujours pas assez de plumes. Il m'en faut " + .@feather_needed + " en tout. Les Lunatic trainent un peu partout sur le champ.";
			close;
		}
		delitem 949,.@feather_needed;
		mes "[La folle de Prontera]";
		mes "Ah ! Voila qui est mieux. Laisse-moi juste... oui, comme ca, et ca...";
		next;
		mes "[La folle de Prontera]";
		mes "Une echarpe pour l'hiver, tissee avec ce que le champ veut bien donner. Les gens rient de moi, tu sais. Ils riront moins quand il fera froid et qu'ils grelotteront sans la leur.";
		close2;
		getexp 13535,0;
		if (getfaction() == 1)
			add_reputation_points(REPUTATION_ORDRE,10);
		else if (getfaction() == 2)
			add_reputation_points(REPUTATION_FORGENOIRES,10);
		else {
			add_reputation_points(REPUTATION_ORDRE,5);
			add_reputation_points(REPUTATION_FORGENOIRES,5);
		}
		completequest 91008;
		end;
	}

	mes "[La folle de Prontera]";
	mes "*Elle fredonne en triant un tas de brindilles, sans lever les yeux.*";
	next;
	mes "[La folle de Prontera]";
	mes "Tu regardes mes mains ? Tout le monde regarde mes mains. Personne ne demande jamais ce qu'elles fabriquent.";
	next;
	mes "[La folle de Prontera]";
	mes "Une echarpe. La plus douce du royaume, si j'y arrive. Mais il me manque des plumes - celles des Lunatic, ici meme, sur le champ.";
	next;
	mes "[La folle de Prontera]";
	mes "" + .@feather_needed + " plumes, pas une de moins. Les Lunatic n'en meurent pas, ils en ont trop. Ca ne te derangerait pas de m'en rapporter, dis ?";
	next;
	switch (select("Pas le temps.", "D'accord, je vais chercher ca.")) {
	case 1:
		mes "[La folle de Prontera]";
		mes "Personne n'a jamais le temps pour une vieille folle et ses fils. Tant pis.";
		close;
	case 2:
		setquest 91008;
		mes "[La folle de Prontera]";
		mes "" + .@feather_needed + " plumes de Lunatic. Reviens quand tu les as, je ne bouge pas d'ici - je n'ai nulle part ou aller, de toute facon.";
		close;
	}
OnQuestIcon:
	if (isbegin_quest(91008) == 2) {
		showevent QTYPE_NONE;
	} else if (isbegin_quest(91008) == 1) {
		if (countitem(949) >= 15) {
			showevent QTYPE_QUEST;
		} else {
			showevent QTYPE_QUEST2;
		}
	} else {
		showevent QTYPE_QUEST;
	}
	end;
}
